Each directive only sets or removes ARIA attributes, kept in step with the library's own classes by a `MutationObserver` that watches `class` and child changes (not the attributes it sets itself, so it never loops):
  ```js
  // src/a11y-attributes.js
  // Attributes on markup that libraries write and that takes none from the app. Each part
  // only sets or removes ARIA attributes, in step with the library's own classes: it adds,
  // moves and draws nothing, and adds no Tab stop or event handler.

  function set(el, name, value) {
      if (value == null || value === false) {
          el.removeAttribute(name);
      } else if (el.getAttribute(name) !== String(value)) {
          el.setAttribute(name, String(value));
      }
  }

  function keepInStep(target, apply) {
      apply();
      const observer = new MutationObserver(apply);
      observer.observe(target, { subtree: true, childList: true, attributes: true, attributeFilter: [ "class" ] });
      return observer;
  }

  /** vue-multiselect: v-multiselect-a11y="{ label?, labelledby?, describedby?, removeLabel?: (text) => string }" */
  export const multiselectA11y = {
      mounted(el, binding) {
          el.__a11yOptions = binding.value ?? {};
          el.__a11yObserver = keepInStep(el, () => {
              const o = el.__a11yOptions;
              const input = el.querySelector(".multiselect__input");
              const box = input ?? el;
              if (input) {
                  // the combobox is the field people type in; the list of options sits beside it
                  set(el, "role", null);
                  set(el, "aria-owns", null);
                  set(input, "role", "combobox");
              }
              set(box, "aria-expanded", el.classList.contains("multiselect--active") ? "true" : "false");
              set(box, "aria-label", o.label);
              set(box, "aria-labelledby", o.labelledby);
              set(box, "aria-describedby", o.describedby);
              const list = el.querySelector(".multiselect__content");
              if (list) {
                  set(list, "aria-labelledby", o.labelledby);
                  set(list, "aria-label", o.labelledby ? null : o.label ?? (input?.labels?.[0]?.textContent.trim() || null));
              }
              for (const li of el.querySelectorAll("li.multiselect__element[role=option]")) {
                  set(li, "aria-selected", li.querySelector(".multiselect__option--selected") ? "true" : null);
              }
              for (const icon of el.querySelectorAll(".multiselect__tag-icon")) {
                  set(icon, "role", "button");
                  if (o.removeLabel) {
                      set(icon, "aria-label", o.removeLabel(icon.previousElementSibling?.textContent.trim() ?? ""));
                  }
              }
          });
      },
      updated(el, binding) {
          el.__a11yOptions = binding.value ?? {};
      },
      unmounted(el) {
          el.__a11yObserver?.disconnect();
      },
  };

  /** v-pagination-3: v-pagination-a11y="{ label, previous, next, page: (n) => string }" */
  export const paginationA11y = {
      mounted(el, binding) {
          el.__a11yObserver = keepInStep(el, () => {
              const o = binding.value ?? {};
              const nav = el.querySelector("nav");
              if (nav) {
                  set(nav, "aria-label", o.label);
              }
              for (const li of el.querySelectorAll(".VuePagination__pagination-item")) {
                  const button = li.querySelector(".page-link");
                  if (!button) {
                      continue;
                  }
                  const text = button.textContent.trim();
                  if (li.classList.contains("VuePagination__pagination-item-prev-page")) {
                      set(button, "aria-label", o.previous);
                  } else if (li.classList.contains("VuePagination__pagination-item-next-page")) {
                      set(button, "aria-label", o.next);
                  } else if (/^\d+$/.test(text)) {
                      set(button, "aria-label", o.page?.(text));
                  }
                  set(button, "aria-current", li.classList.contains("active") ? "page" : null);
              }
          });
      },
      unmounted(el) {
          el.__a11yObserver?.disconnect();
      },
  };

  /** vue-datepicker's field: v-datepicker-a11y="{ label }" */
  export const datepickerA11y = {
      mounted(el, binding) {
          el.__a11yObserver = keepInStep(el, () => {
              const input = el.querySelector("input.dp__input");
              const wrapper = input?.closest("[role=textbox]");
              if (wrapper && wrapper !== input) {
                  // one text field, not one inside another
                  for (const name of [ "role", "aria-label", "aria-multiline", "aria-disabled", "aria-readonly" ]) {
                      set(wrapper, name, null);
                  }
              }
              if (input) {
                  set(input, "aria-label", binding.value?.label);
              }
          });
      },
      unmounted(el) {
          el.__a11yObserver?.disconnect();
      },
  };

  /** vue-prism-editor: v-prism-a11y="{ labelledby }" names the textarea it writes */
  export const prismA11y = {
      mounted(el, binding) {
          el.__a11yObserver = keepInStep(el, () => {
              const textarea = el.querySelector("textarea");
              if (textarea) {
                  set(textarea, "aria-labelledby", binding.value?.labelledby);
              }
          });
      },
      unmounted(el) {
          el.__a11yObserver?.disconnect();
      },
  };

  /**
   * vue-datepicker's menu, which it teleports to <body>: its Cancel and Select are buttons, and
   * in a range time picker each side's controls say which side they set.
   * @param {(key: string, values?: object) => string} t the app's translation function
   * @returns {void}
   */
  export function datepickerMenuA11y(t) {
      keepInStep(document.body, () => {
          for (const action of document.querySelectorAll(".dp__menu .dp__action")) {
              set(action, "role", "button");
          }
          for (const menu of document.querySelectorAll(".dp__menu")) {
              const sides = menu.querySelectorAll(".dp__time_input");
              if (sides.length !== 2) {
                  continue;
              }
              sides.forEach((side, index) => {
                  const which = t(index === 0 ? "timeRangeStart" : "timeRangeEnd");
                  for (const control of side.querySelectorAll("[aria-label]")) {
                      const own = control.dataset.a11yLabel ?? control.getAttribute("aria-label");
                      control.dataset.a11yLabel = own;
                      set(control, "aria-label", t("timeRangeControl", [ which, own ]));
                  }
              });
          }
      });
  }
  ```

### C-45 A semantic sitemap generated from the router, and linked from every page

- **Status:** checked

- **Where:** a new `config/semantic-sitemap.plugin.js`, added to `plugins` in `config/vite.config.js:21`; `meta.sitemap` beside C-5's `meta.screen` on the routes in `src/router.js`; one `<link>` in `index.html:8`
- **Change:** the plugin reads the route table from `src/router.js` itself, as data: it parses the file with Rollup's parser (`rollup/parseAst`, installed with Vite; the route file imports `.vue` components, so it can't be run in Node) and walks the `routes` array's `path`, `children` and `meta.sitemap` literals. A child path that starts with `/` (`/edit/:id` under `/dashboard/:id`) is placed by its full path, as vue-router does. It writes the map into the build (`this.emitFile`, so `dist/semantic-sitemap.json` is served by the server's static `dist`, server/server.js:359) and serves it in dev (`configureServer`):
  ```js
  // config/semantic-sitemap.plugin.js
  import { readFileSync } from "node:fs";
  import { parseAst } from "rollup/parseAst";

  const ROUTER = new URL("../src/router.js", import.meta.url);

  /** The value of a literal AST node; anything else (a component, a function) is undefined. */
  function literal(node) {
      switch (node?.type) {
          case "Literal":
              return node.value;
          case "ArrayExpression":
              return node.elements.map(literal);
          case "ObjectExpression":
              return Object.fromEntries(node.properties.map((p) => [ p.key.name ?? p.key.value, literal(p.value) ]));
          default:
              return undefined;
      }
  }

  /** The route table of src/router.js as plain objects: { path, children, meta } */
  function readRoutes() {
      const ast = parseAst(readFileSync(ROUTER, "utf8"));
      const declaration = ast.body.flatMap((s) => s.declarations ?? []).find((d) => d.id.name === "routes");
      return literal(declaration.init);
  }

  /** Put a page at its segments under the node of the longest path it starts with. */
  function place(node, segments, name) {
      for (const child of node.children ?? []) {
          const own = child.path.split("/");
          if (own.length < segments.length && own.every((s, i) => s === segments[i])) {
              return place(child, segments.slice(own.length), name);
          }
      }
      node.children ??= [];
      node.children.push({ path: segments.join("/"), ...(name ? { name } : {}) });
  }

  /**
   * The sitemap: one node per page (a route with meta.screen), nested by path. A child path that
   * starts with "/" is placed by that full path, as vue-router does; ":id" becomes
   * "{<meta.sitemap.params.id>}"; meta.sitemap === false leaves a page out.
   */
  export function buildSitemap(routes) {
      const root = { path: "/" };
      const params = {};
      const walk = (list, parentPath) => {
          for (const route of list ?? []) {
              const full = route.path.startsWith("/") ? route.path : `${parentPath}/${route.path}`.replace(/\/+$/, "");
              Object.assign(params, route.meta?.sitemap?.params);
              if (route.meta?.screen && route.meta?.sitemap !== false) {
                  const segments = full.replace(/^\//, "").split("/")
                      .map((s) => (s.startsWith(":") ? `{${params[s.slice(1)] ?? s.slice(1)}}` : s));
                  place(root, segments, route.meta.sitemap?.name);
              }
              walk(route.children, full);
          }
      };
      walk(routes, "");
      return root;
  }

  export default function semanticSitemap() {
      const source = () => JSON.stringify(buildSitemap(readRoutes()), null, 4);
      return {
          name: "semantic-sitemap",
          configureServer(server) {
              server.middlewares.use("/semantic-sitemap.json", (req, res) => {
                  res.setHeader("Content-Type", "application/json");
                  res.end(source());
              });
          },
          generateBundle() {
              this.emitFile({ type: "asset", fileName: "semantic-sitemap.json", source: source() });
          },
      };
  }
  ```
